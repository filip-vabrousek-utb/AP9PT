## @AppStorage

```swift
struct UDefaults: View {
    @AppStorage("name") var name: String = "Filip"

    var body: some View {
        VStack {
         Text("Welcome \(name)")
        
         Button("Log in"){
                self.name = "Karel"
            }
        }
    }
}
```


## SceneStorage

```swift
struct StorageView: View {
    // used for state restoration, destroyed on ForceQuit
    @SceneStorage("Saved") var text = ""
    
    var body: some View {
        NavigationView {
            TextEditor(text: $text)
        }
    }
}


```



## @StateObject
* observableObject won't get destroyed, when the View updates


## @ScaledMetric
* scales the text depending on user's Dynamic Type settings

```swift
struct MetricView: View {
    @ScaledMetric var size: CGFloat = 100
    
    var body: some View {
        Image(systemName: "sun.min.fill")
            .resizable()
            .frame(width: size, height: size)
    }
}
```
