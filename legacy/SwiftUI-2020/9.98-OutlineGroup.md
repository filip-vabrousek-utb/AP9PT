## OutlineGroup
* ```outlineGroup``` allows us to nest multiple levels

```swift
struct ContentView: View {
    let data = Archive(name: "Secret",sub: [
    Archive(name: "Sub-Secret", sub: [])
    ])

    var body: some View {
        OutlineGroup(data, children:\.sub){ item in
            Text(item.description)
        }
    }
}
```

```swift
struct Archive: Hashable, Identifiable, CustomStringConvertible {
    var id: Self { self }
    var name: String
    var sub: [Archive]? = nil
    var description: String {
        return name
    }
}
```
