## Toolbar

```swift
struct Styles: View {
    var body: some View {
        NavigationView {
            Text("Hello")
                .padding()
                .toolbar {                       // Compiles without a dot, we don't see anything
                ToolbarItem(placement: .bottomBar) {
                    Button("Bottom"){
                        print("Hello nice!")
                    }
                }
            }
        }
    }
}
```
