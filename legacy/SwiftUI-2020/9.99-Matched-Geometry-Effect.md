## Matched Geometry Effect

```swift
struct MatchedGeometry: View {
 @State var expand = false
 @State var progress: CGFloat = 0.0
 @Namespace var details
 
 var body: some View { 
 VStack { 
 Spacer()

 // minimised View
            VStack(spacing: 35) {
                HStack(spacing: 15){
                    Image("billie").resizable()
                        .frame(width: self.expand ? 250 : 50, height: self.expand ? 250 : 50)
                        .padding(.top, self.expand ? 35 : 0)
                    
                  
                    // hiding the View if its epxanded
                    if !self.expand {
                        Group {
                        VStack(alignment: .leading, spacing: 4){
                            Text("No Time To Die")
                            Text("Billie Eilish").foregroundColor(.pink)
                        }.matchedGeometryEffect(id: "Details", in: self.details)
                        
                        Spacer()
                        
                        
                        Button(action: {
                            print("Wow")
                        }, label:{
                            Image(systemName: "play.fill").foregroundColor(.black)
                               }).matchedGeometryEffect(id: "Play", in: self.details)
                        }
                    }
                }
                
                
                
                
                if self.expand {
                    Group {
                    VStack(alignment: .center, spacing: 4){
                        Text("No Time To Die")
                        Text("Billie Eilish").foregroundColor(.pink)
                    }.matchedGeometryEffect(id: "Details", in: self.details)
                    
                    Spacer()
                    
                    
                        
                        
                      
                        Slider(value: $progress)
                    
                // HStack....................................
                        HStack {
                            Button(action: {
                            print("Wow")
                        }, label:{
                            Image(systemName: "backward.fill")
                                .font(.title)
                                .foregroundColor(.black)
                    })
                            
                            
                            
                            
                            Spacer()
                            
                            
                            Button(action: {
                                    print("Wow")
                                }, label:{
                                    Image(systemName: "play.fill")
                                        .font(.title)
                                        .foregroundColor(.black)
                            }).matchedGeometryEffect(id: "Play", in: self.details)
                            
                            
                            Spacer()
                            
                            
                            Button(action: {
                                    print("Wow")
                                }, label:{
                                    Image(systemName: "forward.fill")
                                        .foregroundColor(.black)
                                        .font(.title)
                            })
                        }.padding(.horizontal)
                        // HStack....................................
                        
                        HStack {
                            Button(action: {}, label: {
                                Image(systemName: "square.and.arrow.down.fill")
                                    .foregroundColor(.black)
                                    .font(.title)
                            })
                            
                            Spacer()
                        }
                    }
                    
                    Spacer()
                }
              
            }.padding(.all)
            .background(Color.white.shadow(radius: 10).onTapGesture {
                withAnimation(Animation.easeOut(duration: 0.4)){
                self.expand.toggle()
                }
            })
        }.background(Color.black
                        .opacity(0.06)
                        .edgesIgnoringSafeArea(.all))
        
    }
}

```
