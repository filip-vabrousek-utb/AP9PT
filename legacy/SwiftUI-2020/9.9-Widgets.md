## Widgets

```swift
import WidgetKit
import SwiftUI
import Intents

@main
struct Filip: Widget {
    public var body: some WidgetConfiguration {
        StaticConfiguration(kind: "com.example.Filip",
                            provider: Provider(),
                            placeholder: Text("Placeholder")) { entry in
                                WidgetView(entry: entry)
        }
    }
}

// 20:00?
struct Provider: TimelineProvider {
    typealias Entry = WidgetEntry
    
    // Current state of the Widget (if we make a network call)
    func snapshot(with context: Context, completion: @escaping (WidgetEntry) -> ()) {
        let date = Date()
        let entry: WidgetEntry
        
        if context.isPreview {
            entry = WidgetEntry(date: date, health: 0.75)
        } else {
            entry = WidgetEntry(date: date, health: BaterryData.currentLevel)
        }
        
        completion(entry)
    }
    
    
    func timeline(with context: Context, completion: @escaping (Timeline<WidgetEntry>) -> ()) {
        UIDevice.current.isBatteryMonitoringEnabled = true
        let entry = WidgetEntry(date: Date(), health: Double(UIDevice.current.batteryLevel))
        let timeline = Timeline(entries: [entry], policy: .atEnd) // update when the previous entry passes
        completion(timeline)
    }
}


struct WidgetView: View {
    var entry: WidgetEntry
    
    var body: some View {
        Text("Battery is at \(Double(UIDevice.current.batteryLevel)) %")
    }
}

struct WidgetEntry: TimelineEntry {
    var date: Date
    var health: Double
}

struct BaterryData  {
    static var currentLevel = Double(UIDevice.current.batteryLevel)
}
```




# Even / Odd week widget
```swift
import WidgetKit
import SwiftUI
import Intents

@main
struct Filip: Widget {
    public var body: some WidgetConfiguration {
        StaticConfiguration(kind: "com.example.Filip",
                            provider: Provider(),
                            placeholder: PlaceholderView()) { entry in
            WidgetView(entry: entry)
                                 
        }.supportedFamilies([.systemSmall]) // only small family
    }
}


struct PlaceholderView: View {
    var body: some View {
        Text("Loading...").widgetURL(URL(string: "www.google.com"))
    }
}


// 20:00?
struct Provider: TimelineProvider {
    typealias Entry = WidgetEntry
    
    // Current state of the Widget (if we make a network call)
    func snapshot(with context: Context, completion: @escaping (WidgetEntry) -> ()) {
        let date = Date()
        let entry: WidgetEntry
    
        entry = WidgetEntry(date: date, parity: "n/a")
        completion(entry)
    }
    
    func timeline(with context: Context, completion: @escaping (Timeline<WidgetEntry>) -> ()) {
       
        let week = Calendar.current.component(.weekOfYear, from: Date())
        let date = week % 2 == 0 ? "Even" : "Odd"
        let entry = WidgetEntry(date: Date(), parity: date)
        let timeline = Timeline(entries: [entry], policy: .atEnd) 
        completion(timeline)
    }
}


struct WidgetView: View {
    var entry: WidgetEntry
    
    var body: some View {
        Text("\(entry.parity) week")
    }
}

struct WidgetEntry: TimelineEntry {
    var date: Date
    var parity: String
}
```
