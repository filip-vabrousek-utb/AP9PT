## Date Formatting inside Text

```swift
 Text(Date(), style: .time)
```

## TextCase
```swift
struct TextStyle: View {
    @State var name:String = ""
    
    var body: some View {
        TextField("Hello", text: $name)
            .textCase(.uppercase)
    }
}
```

## onChange
```swift
 TextField("Hello", text: $name)
            .onChange(of: name) { value in
                print("Value changed \(name)")
            }
```


## ListStyle
```swift
struct ListStyle: View {
    var body: some View {
        List(0..<100){i in
            Text("Row \(i)")
        }.listStyle(InsetGroupedListStyle())
    }
}
```
