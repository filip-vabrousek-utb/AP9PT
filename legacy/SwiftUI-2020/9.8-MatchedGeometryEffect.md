## MatchedGeometryEffect

```swift

struct ContentView: View {
    @Namespace private var animation
    @State private var isZoomed = false
    
    var frame:CGFloat {
        isZoomed ? 300 : 40
    }
    
    var body: some View {
        VStack {
            Spacer()
            
            VStack {
                HStack {
                    RoundedRectangle(cornerRadius: 10)
                        .fill(Color.blue)
                        .frame(width: frame, height: frame)
                        .padding(.top, isZoomed ? 20 : 0)
                    
                    if isZoomed == false {
                        Text("Taylor Swift")
                            .matchedGeometryEffect(id: "Album", in: animation)
                            .font(.headline)
    
                        Spacer()
                    }
                    
                    
                    if isZoomed == true {
                        Text("Taylor Swift")
                            .matchedGeometryEffect(id: "Album", in: animation)
                            .font(.headline)
                            .padding(.bottom, 60)
                        Spacer()
                    }
                     
                }.onTapGesture {
                    withAnimation(.spring()){
                        self.isZoomed.toggle()
                    }
                }
                
                
            }.padding()
            .frame(maxWidth: .infinity)
            .frame(height:isZoomed ? 300 : 60)
            .background(Color(white: 0.9))
            
        }
    }
}


// SOURCE https://www.hackingwithswift.com/quick-start/swiftui/how-to-synchronize-animations-from-one-view-to-another-with-matchedgeometryeffect
```
